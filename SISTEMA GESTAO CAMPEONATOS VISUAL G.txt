algoritmo "TeamUp"
var
   nomes: vetor[1..3,1..4] de caractere
   pontos: vetor[1..3,1..4] de inteiro
   vitorias: vetor[1..3,1..4] de inteiro
   total: vetor[1..3] de inteiro
   corrida: vetor[1..4,1..4] de inteiro
   rodadaFeita: vetor[1..4] de logico
   jogoA: vetor[1..3,1..20] de inteiro
   jogoB: vetor[1..3,1..20] de inteiro
   golsA: vetor[1..3,1..20] de inteiro
   golsB: vetor[1..3,1..20] de inteiro
   tempo1A: vetor[1..20] de inteiro
   tempo1B: vetor[1..20] de inteiro
   tempo2A: vetor[1..20] de inteiro
   tempo2B: vetor[1..20] de inteiro
   penA: vetor[1..20] de inteiro
   penB: vetor[1..20] de inteiro
   tevePenalti: vetor[1..20] de logico
   jogado: vetor[1..3,1..20] de logico
   totalJogos: vetor[1..3] de inteiro
   ordem: vetor[1..4] de inteiro
   campeaoHall: vetor[1..3] de caractere
   pontosHall: vetor[1..3] de inteiro
   salvoHall: vetor[1..3] de logico
   tituloModalidade: vetor[1..3] de caractere
   modalidade, opcao, i, j, r, valor, a, b: inteiro
   numero, x, y, aux, tpA, tpB, pA, pB: inteiro
   vencedores, concluidas: inteiro
   resposta: caractere
   valido, sair, voltar, empate: logico
inicio
   para i de 1 ate 3 faca
      total[i] <- 0
      salvoHall[i] <- FALSO
      campeaoHall[i] <- ""
      pontosHall[i] <- 0
      totalJogos[i] <- 0
      para j de 1 ate 4 faca
         pontos[i,j] <- 0
         vitorias[i,j] <- 0
      fimpara
      para j de 1 ate 20 faca
         jogado[i,j] <- FALSO
      fimpara
   fimpara
   para r de 1 ate 4 faca
      rodadaFeita[r] <- FALSO
      para i de 1 ate 4 faca
         corrida[i,r] <- 0
      fimpara
   fimpara
   tituloModalidade[1] <- "Corrida de carros"
   tituloModalidade[2] <- "Futebol"
   tituloModalidade[3] <- "Basquete"
   sair <- FALSO
   repita
      escreval("")
      escreval("===== TEAMUP - MODALIDADES =====")
      escreval("1 - Corrida de carros")
      escreval("2 - Futebol")
      escreval("3 - Basquete")
      escreval("4 - Hall da Fama")
      escreval("0 - Encerrar programa")
      escreva("Escolha: ")
      leia(modalidade)
      se modalidade = 0 entao
         sair <- VERDADEIRO
      senao
         se modalidade = 4 entao
            escreval("===== HALL DA FAMA TEAMUP =====")
            para i de 1 ate 3 faca
               escreval(tituloModalidade[i])
               se salvoHall[i] entao
                  escreval("CAMPEAO: ", campeaoHall[i])
                  escreval("PONTOS: ", pontosHall[i])
               senao
                  escreval("Campeonato ainda nao finalizado.")
               fimse
               escreval("--------------------------")
            fimpara
         senao
         se (modalidade < 1) ou (modalidade > 3) entao
            escreval("Modalidade invalida.")
         senao
            voltar <- FALSO
            repita
               escreval("")
               escreval("===== MENU DO CAMPEONATO =====")
               se modalidade = 1 entao
                  escreval("CORRIDA DE CARROS")
                  escreval("1 - Cadastrar piloto/equipe")
                  escreval("2 - Listar pilotos/equipes")
                  escreval("3 - Registrar pontos de rodada")
                  escreval("4 - Ranking e ganhador")
                  escreval("5 - Relatorio das 4 rodadas")
               senao
                  se modalidade = 2 entao
                     escreval("FUTEBOL")
                  senao
                     escreval("BASQUETE")
                  fimse
                  escreval("1 - Cadastrar equipe")
                  escreval("2 - Listar equipes")
                  escreval("3 - Agendar partida")
                  escreval("4 - Registrar resultado")
                  escreval("5 - Ranking e ganhador")
                  escreval("6 - Relatorio de partidas")
               fimse
               escreval("7 - Salvar campeao no Hall da Fama")
               escreval("9 - Voltar para modalidades")
               escreval("0 - Encerrar programa")
               escreva("Escolha: ")
               leia(opcao)
               escolha opcao
                  caso 1
                     se salvoHall[modalidade] entao
                        escreval("Campeonato finalizado. Cadastro/alteracao bloqueado.")
                     senao
                     concluidas <- 0
                     se modalidade = 1 entao
                        para r de 1 ate 4 faca
                           se rodadaFeita[r] entao
                              concluidas <- concluidas + 1
                           fimse
                        fimpara
                     fimse
                     se concluidas > 0 entao
                        escreval("Cadastros encerrados apos iniciar as rodadas.")
                     senao
                     se total[modalidade] < 4 entao
                        total[modalidade] <- total[modalidade] + 1
                        escreva("Nome do piloto/equipe: ")
                        leia(nomes[modalidade,total[modalidade]])
                        escreval("Cadastrado! ID: ", total[modalidade])
                     senao
                        escreval("Limite de 4 participantes.")
                     fimse
                     fimse
                     fimse
                  caso 2
                     escreval("===== CADASTRADOS =====")
                     se total[modalidade] = 0 entao
                        escreval("Nenhum cadastro.")
                     fimse
                     para i de 1 ate total[modalidade] faca
                        escreval(i, " - ", nomes[modalidade,i])
                     fimpara
                  caso 3
                     se salvoHall[modalidade] entao
                        escreval("Campeonato finalizado. Cadastro/alteracao bloqueado.")
                     senao
                     se modalidade = 1 entao
                        se total[1] = 0 entao
                           escreval("Cadastre pilotos primeiro.")
                        senao
                           escreva("Rodada (1 a 4): ")
                           leia(r)
                           se (r < 1) ou (r > 4) entao
                              escreval("Rodada invalida.")
                           senao
                              se rodadaFeita[r] entao
                                 escreval("Rodada ja registrada.")
                              senao
                                 valido <- VERDADEIRO
                                 para i de 1 ate total[1] faca
                                    escreva(nomes[1,i], " - pontos: ")
                                    leia(valor)
                                    se valor < 0 entao
                                       valido <- FALSO
                                    senao
                                       corrida[i,r] <- valor
                                    fimse
                                 fimpara
                                 se valido entao
                                    rodadaFeita[r] <- VERDADEIRO
                                    para i de 1 ate total[1] faca
                                       pontos[1,i] <- pontos[1,i] + corrida[i,r]
                                    fimpara
                                    escreval("Rodada registrada!")
                                 senao
                                    escreval("Pontos negativos: rodada nao salva.")
                                 fimse
                              fimse
                           fimse
                        fimse
                     senao
                        se total[modalidade] < 2 entao
                           escreval("Cadastre duas equipes primeiro.")
                        senao
                           se totalJogos[modalidade] >= 20 entao
                              escreval("Limite de 20 partidas.")
                           senao
                              para i de 1 ate total[modalidade] faca
                                 escreval(i, " - ", nomes[modalidade,i])
                              fimpara
                              escreva("ID equipe A: ")
                              leia(a)
                              escreva("ID equipe B: ")
                              leia(b)
                              se (a < 1) ou (a > total[modalidade]) ou (b < 1) ou (b > total[modalidade]) ou (a = b) entao
                                 escreval("Equipes invalidas.")
                              senao
                                 totalJogos[modalidade] <- totalJogos[modalidade] + 1
                                 numero <- totalJogos[modalidade]
                                 jogoA[modalidade,numero] <- a
                                 jogoB[modalidade,numero] <- b
                                 jogado[modalidade,numero] <- FALSO
                                 escreval("Partida ", numero, " agendada!")
                              fimse
                           fimse
                        fimse
                     fimse
                     fimse
                  caso 4
                     se modalidade = 1 entao
                        escreval("===== RANKING CORRIDA =====")
                        para i de 1 ate total[1] faca
                           ordem[i] <- i
                        fimpara
                        para i de 1 ate total[1] faca
                           para j de i+1 ate total[1] faca
                              se pontos[1,ordem[j]] > pontos[1,ordem[i]] entao
                                 aux <- ordem[i]
                                 ordem[i] <- ordem[j]
                                 ordem[j] <- aux
                              fimse
                           fimpara
                        fimpara
                        para i de 1 ate total[1] faca
                           escreval(i, " lugar: ", nomes[1,ordem[i]], " - ", pontos[1,ordem[i]], " pontos")
                        fimpara
                        concluidas <- 0
                        para r de 1 ate 4 faca
                           se rodadaFeita[r] entao
                              concluidas <- concluidas + 1
                           fimse
                        fimpara
                        se total[1] = 0 entao
                           escreval("Cadastre participantes primeiro.")
                        senao
                           se concluidas < 4 entao
                              escreval("Ranking parcial: ", concluidas, " de 4 rodadas.")
                           senao
                              vencedores <- 0
                              para i de 1 ate total[1] faca
                                 se pontos[1,i] = pontos[1,ordem[1]] entao
                                    vencedores <- vencedores + 1
                                 fimse
                              fimpara
                              se vencedores = 1 entao
                                 escreval("CAMPEAO: ", nomes[1,ordem[1]])
                              senao
                                 escreval("EMPATE NO PRIMEIRO LUGAR!")
                                 escreval("Confira os pontos dos empatados.")
                              fimse
                           fimse
                        fimse
                     senao
                        se salvoHall[modalidade] entao
                           escreval("Campeonato finalizado. Resultados bloqueados.")
                        senao
                        se totalJogos[modalidade] = 0 entao
                           escreval("Agende uma partida primeiro.")
                        senao
                           para i de 1 ate totalJogos[modalidade] faca
                              escreva(i, " - ", nomes[modalidade,jogoA[modalidade,i]], " x ", nomes[modalidade,jogoB[modalidade,i]])
                              se jogado[modalidade,i] entao
                                 escreval(" - FINALIZADA")
                              senao
                                 escreval(" - PENDENTE")
                              fimse
                           fimpara
                           escreva("Numero da partida: ")
                           leia(numero)
                           se (numero < 1) ou (numero > totalJogos[modalidade]) entao
                              escreval("Partida invalida.")
                           senao
                              se jogado[modalidade,numero] entao
                                 escreval("Resultado ja registrado.")
                              senao
                                 a <- jogoA[modalidade,numero]
                                 b <- jogoB[modalidade,numero]
                                 valido <- VERDADEIRO
                                 se modalidade = 2 entao
                                    escreval("===== FUTEBOL: DOIS TEMPOS =====")
                                    escreva("Gols 1 tempo - ", nomes[2,a], ": ")
                                    leia(x)
                                    escreva("Gols 1 tempo - ", nomes[2,b], ": ")
                                    leia(y)
                                    escreva("Gols 2 tempo - ", nomes[2,a], ": ")
                                    leia(tpA)
                                    escreva("Gols 2 tempo - ", nomes[2,b], ": ")
                                    leia(tpB)
                                    se (x < 0) ou (y < 0) ou (tpA < 0) ou (tpB < 0) entao
                                       valido <- FALSO
                                    senao
                                       tempo1A[numero] <- x
                                       tempo1B[numero] <- y
                                       tempo2A[numero] <- tpA
                                       tempo2B[numero] <- tpB
                                       x <- x + tpA
                                       y <- y + tpB
                                       escreval("Placar final: ", x, " x ", y)
                                       tevePenalti[numero] <- FALSO
                                       se x = y entao
                                          escreva("Houve disputa de penaltis? (S/N): ")
                                          leia(resposta)
                                          se (resposta = "S") ou (resposta = "s") entao
                                             tevePenalti[numero] <- VERDADEIRO
                                             escreva("Penaltis de ", nomes[2,a], ": ")
                                             leia(pA)
                                             escreva("Penaltis de ", nomes[2,b], ": ")
                                             leia(pB)
                                             se (pA < 0) ou (pB < 0) ou (pA = pB) entao
                                                valido <- FALSO
                                                escreval("Penaltis invalidos: devem definir um vencedor.")
                                             senao
                                                penA[numero] <- pA
                                                penB[numero] <- pB
                                             fimse
                                          fimse
                                       senao
                                          escreval("Sem penaltis: houve vencedor no tempo normal.")
                                       fimse
                                    fimse
                                 senao
                                    escreva("Pontos de ", nomes[3,a], ": ")
                                    leia(x)
                                    escreva("Pontos de ", nomes[3,b], ": ")
                                    leia(y)
                                    se (x < 0) ou (y < 0) ou (x = y) entao
                                       valido <- FALSO
                                       escreval("Basquete exige placar valido sem empate.")
                                    fimse
                                 fimse
                                 se valido entao
                                    golsA[modalidade,numero] <- x
                                    golsB[modalidade,numero] <- y
                                    jogado[modalidade,numero] <- VERDADEIRO
                                    se x > y entao
                                       se modalidade = 2 entao
                                          pontos[2,a] <- pontos[2,a] + 3
                                       senao
                                          pontos[3,a] <- pontos[3,a] + 2
                                       fimse
                                       vitorias[modalidade,a] <- vitorias[modalidade,a] + 1
                                    senao
                                       se y > x entao
                                          se modalidade = 2 entao
                                             pontos[2,b] <- pontos[2,b] + 3
                                          senao
                                             pontos[3,b] <- pontos[3,b] + 2
                                          fimse
                                          vitorias[modalidade,b] <- vitorias[modalidade,b] + 1
                                       senao
                                          pontos[2,a] <- pontos[2,a] + 1
                                          pontos[2,b] <- pontos[2,b] + 1
                                       fimse
                                    fimse
                                    escreval("Resultado registrado!")
                                    se (modalidade = 2) e tevePenalti[numero] entao
                                       se penA[numero] > penB[numero] entao
                                          escreval("Vencedor nos penaltis: ", nomes[2,a])
                                       senao
                                          escreval("Vencedor nos penaltis: ", nomes[2,b])
                                       fimse
                                       escreval("Pontos da tabela: empate no tempo normal (1 para cada).")
                                    fimse
                                 fimse
                              fimse
                           fimse
                        fimse
                        fimse
                     fimse
                  caso 5
                     se modalidade = 1 entao
                        escreval("===== RELATORIO CORRIDA =====")
                        para r de 1 ate 4 faca
                           se rodadaFeita[r] entao
                              escreval("Rodada ", r, ": concluida")
                           senao
                              escreval("Rodada ", r, ": pendente")
                           fimse
                        fimpara
                        para i de 1 ate total[1] faca
                           escreval("Piloto/equipe: ", nomes[1,i])
                           para r de 1 ate 4 faca
                              escreval("Rodada ", r, ": ", corrida[i,r], " pontos")
                           fimpara
                           escreval("TOTAL: ", pontos[1,i], " pontos")
                        fimpara
                        escreval("Use a opcao 4 para ver o campeao.")
                     senao
                        escreval("===== RANKING =====")
                        para i de 1 ate total[modalidade] faca
                           ordem[i] <- i
                        fimpara
                        para i de 1 ate total[modalidade] faca
                           para j de i+1 ate total[modalidade] faca
                              se pontos[modalidade,ordem[j]] > pontos[modalidade,ordem[i]] entao
                                 aux <- ordem[i]
                                 ordem[i] <- ordem[j]
                                 ordem[j] <- aux
                              fimse
                           fimpara
                        fimpara
                        para i de 1 ate total[modalidade] faca
                           escreval(i, " lugar: ", nomes[modalidade,ordem[i]], " - ", pontos[modalidade,ordem[i]], " pontos")
                        fimpara
                        concluidas <- 0
                        para i de 1 ate totalJogos[modalidade] faca
                           se jogado[modalidade,i] entao
                              concluidas <- concluidas + 1
                           fimse
                        fimpara
                        se (total[modalidade] = 0) ou (totalJogos[modalidade] = 0) entao
                           escreval("Ainda nao ha campeonato com partidas.")
                        senao
                           se concluidas < totalJogos[modalidade] entao
                              escreval("Ranking parcial: existem partidas pendentes.")
                           senao
                              vencedores <- 0
                              para i de 1 ate total[modalidade] faca
                                 se pontos[modalidade,i] = pontos[modalidade,ordem[1]] entao
                                    vencedores <- vencedores + 1
                                 fimse
                              fimpara
                              se vencedores = 1 entao
                                 escreval("CAMPEAO POR PONTOS: ", nomes[modalidade,ordem[1]])
                              senao
                                 escreval("EMPATE NA LIDERANCA!")
                                 escreval("E necessario definir um desempate.")
                              fimse
                           fimse
                        fimse
                     fimse
                  caso 6
                     se modalidade = 1 entao
                        escreval("Opcao invalida para corrida.")
                     senao
                        escreval("===== RELATORIO DE PARTIDAS =====")
                        para i de 1 ate totalJogos[modalidade] faca
                           escreva("Partida ", i, ": ", nomes[modalidade,jogoA[modalidade,i]], " x ", nomes[modalidade,jogoB[modalidade,i]])
                           se jogado[modalidade,i] entao
                              escreval(" | ", golsA[modalidade,i], " x ", golsB[modalidade,i])
                              se modalidade = 2 entao
                                 escreval("  1 tempo: ", tempo1A[i], " x ", tempo1B[i])
                                 escreval("  2 tempo: ", tempo2A[i], " x ", tempo2B[i])
                                 se tevePenalti[i] entao
                                    escreval("  Penaltis: ", penA[i], " x ", penB[i])
                                 senao
                                    escreval("  Sem disputa de penaltis")
                                 fimse
                              fimse
                           senao
                              escreval(" | PENDENTE")
                           fimse
                        fimpara
                        escreval("===== PONTUACAO =====")
                        para i de 1 ate total[modalidade] faca
                           escreval(nomes[modalidade,i], ": ", pontos[modalidade,i], " pontos")
                        fimpara
                     fimse
                  caso 7
                     se salvoHall[modalidade] entao
                        escreval("Campeao ja registrado no Hall da Fama.")
                     senao
                        valido <- VERDADEIRO
                        se total[modalidade] < 2 entao
                           escreval("Cadastre pelo menos 2 participantes.")
                           valido <- FALSO
                        senao
                           se modalidade = 1 entao
                              para r de 1 ate 4 faca
                                 se nao rodadaFeita[r] entao
                                    valido <- FALSO
                                 fimse
                              fimpara
                              se nao valido entao
                                 escreval("Registre as 4 rodadas antes de finalizar.")
                              fimse
                           senao
                              se totalJogos[modalidade] = 0 entao
                                 valido <- FALSO
                              fimse
                              para i de 1 ate totalJogos[modalidade] faca
                                 se nao jogado[modalidade,i] entao
                                    valido <- FALSO
                                 fimse
                              fimpara
                              se nao valido entao
                                 escreval("Registre todas as partidas agendadas.")
                              fimse
                           fimse
                        fimse
                        se valido entao
                           a <- 1
                           empate <- FALSO
                           para i de 2 ate total[modalidade] faca
                              se pontos[modalidade,i] > pontos[modalidade,a] entao
                                 a <- i
                                 empate <- FALSO
                              senao
                                 se pontos[modalidade,i] = pontos[modalidade,a] entao
                                    empate <- VERDADEIRO
                                 fimse
                              fimse
                           fimpara
                           vencedores <- 0
                           para i de 1 ate total[modalidade] faca
                              se pontos[modalidade,i] = pontos[modalidade,a] entao
                                 vencedores <- vencedores + 1
                              fimse
                           fimpara
                           se vencedores > 1 entao
                              escreval("Empate na lideranca. Resolva antes de salvar.")
                           senao
                              campeaoHall[modalidade] <- nomes[modalidade,a]
                              pontosHall[modalidade] <- pontos[modalidade,a]
                              salvoHall[modalidade] <- VERDADEIRO
                              escreval("CAMPEAO SALVO NO HALL DA FAMA!")
                              escreval(campeaoHall[modalidade], " - ", pontosHall[modalidade], " pontos")
                           fimse
                        fimse
                     fimse
                  caso 9
                     voltar <- VERDADEIRO
                  caso 0
                     sair <- VERDADEIRO
                     voltar <- VERDADEIRO
                  outrocaso
                     escreval("Opcao invalida.")
               fimescolha
            ate voltar
         fimse
         fimse
      fimse
   ate sair
   escreval("TeamUp encerrado. Ate logo!")
fimalgoritmo
